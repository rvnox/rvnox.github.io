加载字体

在新版本的 fyne 框架中，已经很少遇到 GUI 中显示中文乱码的问题了，但是在老版本中，依然存在 GUI 中显示中文乱码的问题，那么我们可以将字体打包到 exe 中来解决该问题。

首先，我们可以先准备一个字体放入和 main.go 的相同路径下，如：微软雅黑（wryh1.ttf）。

把字体打包到 bundle.go 文件中。然后在和 main.go 同目录创建 theme.go 文件，内容如下：

package main

import (
   "image/color"

   "fyne.io/fyne/v2"
   "fyne.io/fyne/v2/theme"
)

type myTheme struct{}

var _ fyne.Theme = (*myTheme)(nil)

// return bundled font resource
func (*myTheme) Font(s fyne.TextStyle) fyne.Resource {
   if s.Monospace {
      return theme.DefaultTheme().Font(s)
   }
   if s.Bold {
      if s.Italic {
         return theme.DefaultTheme().Font(s)
      }
      return resourceWryh1Ttf
   }
   if s.Italic {
      return theme.DefaultTheme().Font(s)
   }
   return resourceWryh1Ttf

}

func (*myTheme) Color(n fyne.ThemeColorName, v fyne.ThemeVariant) color.Color {
   return theme.DefaultTheme().Color(n, v)
}

func (*myTheme) Icon(n fyne.ThemeIconName) fyne.Resource {
   return theme.DefaultTheme().Icon(n)
}

func (*myTheme) Size(n fyne.ThemeSizeName) float32 {
   return theme.DefaultTheme().Size(n)
}

Fyne 框架实战：共享盘添加助手

因领导需在公司会议室中使用个人共享盘进行临时文件传输，所以需要写一个个人共享盘的添加助手

思路：通过重启 workstation 服务，然后利用 net use 命令来实现个人共享盘的添加

注意点：

  • 普通域用户无管理员权限，而重启 workstation 服务需要管理员权限， 可以利用 runasspc 提权到预管理员权限重启服务

程序大致思路：

程序源码：

package main

import (
   "fmt"
   "github.com/fsnotify/fsnotify"
   "golang.org/x/text/encoding/simplifiedchinese"
   "io"
   "log"
   "os"
   "os/exec"
   "path/filepath"
   "strings"
   "syscall"
   "time"

   "fyne.io/fyne/v2"
   "fyne.io/fyne/v2/app"
   "fyne.io/fyne/v2/canvas"
   "fyne.io/fyne/v2/container"
   "fyne.io/fyne/v2/dialog"
   "fyne.io/fyne/v2/widget"
)

// Charset 定义字符集类型
type Charset string

// 常见字符集
const (
   UTF8    = Charset("UTF-8")
   GB18030 = Charset("GB18030")
)

// ConvertByte2String 函数用于将字节数组转换为字符串，根据指定的字符集（GB18030 或 UTF-8）
func ConvertByte2String(byte []byte, charset Charset) string {
   var str string
   switch charset {
   case GB18030:
   	// 使用 GB18030 解码字节数组
   	decodeBytes, _ := simplifiedchinese.GB18030.NewDecoder().Bytes(byte)
   	str = string(decodeBytes)
   case UTF8:
   	fallthrough
   default:
   	str = string(byte)
   }
   return str
}

func CopyExeToCDrive() error {
   // 获取当前程序所在目录
   exePath, err := os.Executable()
   if err != nil {
   	return fmt.Errorf("无法获取当前路径: %w", err)
   }
   baseDir := filepath.Dir(exePath)

   // 构造源文件和目标文件路径
   src := filepath.Join(baseDir, "重启共享服务.exe")
   dst := "D:\\重启共享服务.exe"

   // 打开源文件
   sourceFile, err := os.Open(src)
   if err != nil {
   	return fmt.Errorf("无法打开源文件 %s: %w", src, err)
   }
   defer sourceFile.Close()

   // 创建目标文件
   destFile, err := os.Create(dst)
   if err != nil {
   	return fmt.Errorf("无法创建目标文件 %s: %w", dst, err)
   }
   defer destFile.Close()

   // 拷贝内容
   _, err = io.Copy(destFile, sourceFile)
   if err != nil {
   	return fmt.Errorf("拷贝失败: %w", err)
   }

   fmt.Println("重启共享服务.exe 已成功拷贝到 D:\\")
   return nil
}

// 定义重启网络服务的函数
func restartNetworkService(workstationPath string) bool {
   // 执行命令
   RunCmd := exec.Command("cmd", "/c", "run.exe")
   RunCmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
   Runout, err := RunCmd.CombinedOutput()
   if err != nil {
   	fmt.Println("执行失败:", err)
   }
   convertedOut := ConvertByte2String(Runout, GB18030)
   fmt.Println("命令输出:", convertedOut)

   // 启动监听
   dir := filepath.Dir(workstationPath)
   fileName := filepath.Base(workstationPath)

   watcher, err := fsnotify.NewWatcher()
   if err != nil {
   	log.Println("创建 watcher 失败:", err)
   	return false
   }
   defer watcher.Close()

   err = watcher.Add(dir)
   if err != nil {
   	log.Println("添加监听失败:", err)
   	return false
   }

   timer := time.NewTimer(10 * time.Second)
   defer timer.Stop()


   if _, err := os.Stat(workstationPath); err == nil {
   	fmt.Println("重启workstation服务成功（文件已存在）")
   	return true
   }

   for {
   	select {
   	case event := <-watcher.Events:
   		if event.Op&fsnotify.Create == fsnotify.Create && filepath.Base(event.Name) == fileName {
   			fmt.Println("重启workstation服务成功（检测到文件创建）")
   			return true
   		}
   	case err := <-watcher.Errors:
   		log.Println("监听错误:", err)
   	case <-timer.C:
   		fmt.Println("监听超时，未检测到文件创建")
   		return false
   	}
   }
}

func main() {
   a := app.New()
   a.SetIcon(resourceLogo4Ico) //  设置任务栏图标和窗口左上角图标
   w := a.NewWindow("会议室个人共享登录助手_V1")

   // 固定大小禁止缩放
   w.Resize(fyne.NewSize(500, 300))
   w.SetFixedSize(true)

   img := canvas.NewImageFromResource(resourceLogo3Png)
   img.FillMode = canvas.ImageFillContain
   img.SetMinSize(fyne.NewSize(500, 100))

   shareEntry := widget.NewEntry()
   shareEntry.SetPlaceHolder("输入个人共享盘名称，如: 小黑$")

   userEntry := widget.NewEntry()
   userEntry.SetPlaceHolder("输入个人电脑账号，如: yxtest")

   passEntry := widget.NewPasswordEntry()
   passEntry.SetPlaceHolder("输入个人电脑密码，如: 123456")

   statusLabel := widget.NewLabel("")

   loginBtn := widget.NewButton("点击登录共享盘", func() {
   	share := strings.TrimSpace(shareEntry.Text)
   	user := strings.TrimSpace(userEntry.Text)
   	pass := strings.TrimSpace(passEntry.Text)

   	if share == "" || user == "" || pass == "" {
   		statusLabel.SetText("❌ 请填写完整信息")
   		return
   	}

   	// 显示模态进度条
   	progress := dialog.NewProgressInfinite("PersonalShare", "正在登录中，请耐心等待...", w)
   	progress.Show()

   	go func() {
   		defer progress.Hide()

   		// 删除 I: 映射
   		deleteCmd := exec.Command("net", "use", "I:", "/delete", "/y")
   		deleteCmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
   		_ = deleteCmd.Run() // 忽略错误，也可以打印调试信息

   		if err := CopyExeToCDrive(); err != nil {
   			fmt.Println("错误:", err)
   		}

   		time.Sleep(1 * time.Second) // 等待片刻，避免过快切换

   		workstationPath := `D:\workstation_restart_success`
   		err := os.Remove(workstationPath)
   		if err != nil {
   			fmt.Println("删除文件失败:", err)
   		} else {
   			fmt.Println("workstation_restart_success 文件已成功删除")
   		}

   
   		if !restartNetworkService(workstationPath) {
   			fmt.Println("第一次重启失败，尝试第二次...")
   			if !restartNetworkService(workstationPath) {
   				msg := fmt.Sprintf("❌ 网络共享服务启动失败，请尝试再次登录 ！")
   				statusLabel.SetText(msg)
   				return
   			}
   		}

   		time.Sleep(1 * time.Second) // 等待片刻

   		// 执行 net use P: \\10.64.110.7\共享盘名称 /user:zg\账号 密码
   		path := fmt.Sprintf(`\\10.64.110.7\%s`, share)
   		username := fmt.Sprintf(`zg\%s`, user)
   		var convertedout string
   		maxRetries := 2

   		for i := 0; i < maxRetries; i++ {
   			cmd := exec.Command("net", "use", "I:", path, "/user:"+username, pass, "/persistent:no")
   			cmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
   			out, _ := cmd.CombinedOutput()
   			convertedout = ConvertByte2String(out, GB18030)
   			fmt.Println("命令输出:", convertedout)

   			if !strings.Contains(convertedout, "找不到网络名") {
   				// 成功或不是“找不到网络名”的错误，就跳出循环
   				break
   			}
   			time.Sleep(2 * time.Second) // 等待片刻
   		}

   		// 判断输出是否包含“成功”
   		if strings.Contains(convertedout, "命令成功完成") {
   			// 刷新任务管理器
   			restartExplorerCmd := exec.Command("cmd", "/c", "start", "I:")
   			restartExplorerCmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
   			_ = restartExplorerCmd.Run() // 忽略返回值和错误
   			statusLabel.SetText("✅ 登录成功，已打开相关共享盘 ！")
   		} else if strings.Contains(convertedout, "找不到网络名") {
   			statusLabel.SetText("❌ 登录失败，请检查网络服务和共享盘名称并再次尝试 ！")
   		} else if strings.Contains(convertedout, "密码不正确") {
   			statusLabel.SetText("❌ 登录失败，请检查电脑账号和密码是否正确并再次尝试 ！")
   		} else if strings.Contains(convertedout, "用户不得从此工作站登录") {
   			statusLabel.SetText("❌ 登录失败，无会议室个人共享登录权限 ！")
   		} else {
   			logFilePath := `D:\share_error.log`
   			_ = os.WriteFile(logFilePath, []byte(convertedout), 0644)
   			msg := fmt.Sprintf("❌ 登录失败，请分析D盘下错误日志 ！")
   			statusLabel.SetText(msg)
   		}
   	}()
   })

   form := container.NewVBox(
   	shareEntry,
   	userEntry,
   	passEntry,
   	loginBtn,
   	statusLabel,
   )

   content := container.NewVBox(img, form)
   w.SetContent(content)
   w.CenterOnScreen()
   w.ShowAndRun()
}

golang 中调用 cmd 执行命令的时候，会存弹出 cmd 的黑窗，可以通过如下方式隐藏

额外补充：golang 中以管理员运行 exe

要使程序在运行时始终需要管理员权限，你需要在程序的manifest文件中设置相关权限要求

首先，创建一个名为app.manifest的 XML 文件，内容如下：

<?xml version="1.0" encoding="utf-8"?>
<assembly xmlns="urn:schemas-microsoft-com:asm.v1" manifestVersion="1.0">
 <trustInfo xmlns="urn:schemas-microsoft-com:asm.v3">
   <security>
     <requestedPrivileges>
       <requestedExecutionLevel level="requireAdministrator" uiAccess="false" />
     </requestedPrivileges>
   </security>
 </trustInfo>
</assembly>

然后，使用 mt.exe（Microsoft Manifest Tool）来将 app.manifest 文件嵌入到最终的 exe 文件中。

mt.exe -manifest app.manifest -outputresource:yourprogram.exe;1

mt.exe 会将 app.manifest 文件嵌入到 yourprogram.exe 中，确保程序运行的时候必须要管理员权限。

额外补充：golang 中为 exe 添加图标

我们可以使用 rsrc 工具来嵌入图标，当然除了用 mt.exe 外，我们也可以使用 rsrc 嵌入 app.manifest 来保证程序运行的时候具有管理员权限。

把 app.ico 作为程序的图标：

首先安装 rsrc

go install github.com/akavel/rsrc@latest

生成 rsrc.syso 文件，rsrc.syso 文件中包含了 app.ico 图标

rsrc -ico app.ico -o rsrc.syso

运行 go build 编译程序，编译的时候，rsrc.syso 会自动和 golang 程序合并，生成包含图标的 exe 文件。

go build

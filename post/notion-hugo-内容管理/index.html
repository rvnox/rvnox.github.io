Notion-Site 源码修改

问题一：图片处理逻辑修改

导致问题的原因无非就两个：

  • Notion 数据库中原来的数据就存在问题，所以通过 notion-site 读取后自然也存在问题。

  • Notion 数据库中原数据正常，但 notion-site 的读取或保存图片逻辑存在问题。

我们先来看第一个，我们通过如下 golang 代码直接读取 notion 数据库中数据：

package main

import (
   "context"
   "fmt"
   "log"

   "github.com/jomei/notionapi"
)

func main() {

   client := notionapi.NewClient("ntn_i7326002xxxxxxxxxxxxxx") // token

   dbID := notionapi.DatabaseID("29d1b4a9exxxxxxxxxxxxxxxxxxx") // 数据库 ID

   // 查询数据库
   query := &notionapi.DatabaseQueryRequest{}
   dbResp, err := client.Database.Query(context.Background(), dbID, query)
   if err != nil {
   	log.Fatal(err)
   }

   fmt.Printf("Total pages: %d\n", len(dbResp.Results))

   for _, page := range dbResp.Results {
   	fmt.Println("Page ID:", page.ID)

   	// 打印页面属性
   	for propName, propVal := range page.Properties {
   		fmt.Printf("  %s: %+v\n", propName, propVal)
   	}

   	// 获取页面 Block（正文）
   	blocksResp, err := client.Block.GetChildren(context.Background(), notionapi.BlockID(page.ID), nil)
   	if err != nil {
   		log.Printf("Failed to get blocks for page %s: %v\n", page.ID, err)
   		continue
   	}

   	// 遍历 Block
   	for _, b := range blocksResp.Results {
   		switch block := b.(type) {
   		case *notionapi.ImageBlock:
   			if block.Image.File != nil {
   				fmt.Println("  Image (File) URL:", block.Image.File.URL)
   			}
   			if block.Image.External != nil {
   				fmt.Println("  Image (External) URL:", block.Image.External.URL)
   			}
   		case *notionapi.FileBlock:
   			if block.File.File != nil {
   				fmt.Println("  File URL:", block.File.File.URL)
   			}
   			if block.File.External != nil {
   				fmt.Println("  External File URL:", block.File.External.URL)
   			}
   		}
   	}

   	fmt.Println("-----------")
   }
}

我们找到有关图片保存的函数 saveTo 函数，如下：

对我们自己搭建的数据库，由于图片 URL 的差异，notion-site 在处理图片时，依据上面的处理逻辑，所有图片的 filename 会完全相同，这样就会覆盖导致前面的问题。

我们修改 filename 处理逻辑，将 saveTo 函数修改为如下逻辑：

func (files *Files) saveTo(reader io.Reader, rawURL, distDir string) (string, error) {
   u, err := url.Parse(rawURL)
   if err != nil {
   	return "", fmt.Errorf("malformed url: %s", err)
   }

   // gen file name: 始终使用倒数第二段 + UUID
   splitPaths := strings.Split(u.Path, "/")
   if len(splitPaths) < 2 {
   	return "", fmt.Errorf("url path too short: %s", u.Path)
   }
   baseName := splitPaths[len(splitPaths)-2]
   ext := filepath.Ext(splitPaths[len(splitPaths)-1])
   if ext == "" {
   	ext = ".png" // 防止没有扩展名
   }

   if err := os.MkdirAll(distDir, 0755); err != nil {
   	return "", fmt.Errorf("%s: %s", distDir, err)
   }

   // 拼接文件名：倒数第二段 + UUID + 扩展名
   filename := fmt.Sprintf("%s_%s%s", baseName, uuid.NewString(), ext)
   out, err := os.Create(filepath.Join(distDir, filename))
   if err != nil {
   	return "", fmt.Errorf("couldn't create image file: %s", err)
   }
   defer out.Close()

   _, err = io.Copy(out, reader)
   return filename, err
}

其中， 文件内容如下：

name: build
on:
 schedule:
   - cron: "0 16 * * *"
   - cron: "0 4 * * *"
 workflow_dispatch:
jobs:
 sync:
   name: Sync from notion
   runs-on: ubuntu-latest
   steps:
     - name: checkout
       uses: actions/checkout@v3
       with:
         submodules: true
         fetch-depth: 0
     - name: Ensure write permissions
       run: |
         if [ -d "./content" ]; then
           chmod -R u+w ./content
         else
           echo "Directory ./content does not exist. Skipping chmod."
         fi
     - name: Force sync with origin/main (preserve README.md)
       run: |
         # 🧩 临时备份 README.md
         mkdir -p /tmp/backup
         if [ -f "./content/readme.md" ]; then
           cp ./content/readme.md /tmp/backup/
           echo "✅ 已备份 README.md"
         else
           echo "ℹ️ 未找到 readme.md，无需备份"
         fi

         # 🧩 强制同步代码
         git fetch origin
         git reset --hard origin/main
         git clean -fd

         # 🧩 恢复 README.md
         if [ -f "/tmp/backup/readme.md" ]; then
           mkdir -p ./content
           cp /tmp/backup/readme.md ./content/
           echo "✅ 已恢复 README.md"
         else
           echo "ℹ️ 无 README.md 需要恢复"
         fi       
     - name: Clean the old post
       run: |
         find ./content -mindepth 1 -not -name 'readme.md' -exec rm -rf {} +
         echo "Content directory cleaned, README.md preserved."
     - name: notion-site
       uses: rvnox/notion-site@main
       env:
         NOTION_SECRET : ${{ secrets.NOTION_SECRET }}
     - name: Check if there are any changes
       id: verify_diff
       run: |
         git diff --quiet . || echo "changed=true" >> $GITHUB_OUTPUT
         
     - name: Set up Git
       run: |
         git config user.email "action@github.com"
         git config user.name "GitHub Actions"

     - name: Commit new results
       run: |
         git add --all
         git commit -m "New results" || echo "No changes to commit"
     
     - name: Push changes
       id: push_changes
       run: |
         git push origin main || echo "Push failed or no changes to push"
         echo "push_success=true" >> $GITHUB_OUTPUT

     - name: Setup Hugo
       if: steps.push_changes.outputs.push_success == 'true'
       uses: peaceiris/actions-hugo@v2
       with:
         hugo-version: '0.121.1'
         extended: true

     - uses: actions/cache@v4
       if: steps.push_changes.outputs.push_success == 'true'
       with:
         path: /tmp/hugo_cache
         key: ${{ runner.os }}-hugomod-${{ hashFiles('**/go.sum') }}
         restore-keys: |
           ${{ runner.os }}-hugomod-

     - name: Build
       if: steps.push_changes.outputs.push_success == 'true'
       run: |
         if [ -f "./content/blogs.json" ]; then
           echo "检测到 content/blogs.json，准备删除..."
           rm -f ./content/blogs.json
     
           # 再次检查是否删除成功
           if [ ! -f "./content/blogs.json" ]; then
             echo "✅ blogs.json 删除成功。"
           else
             echo "❌ 删除 blogs.json 失败，请检查权限或路径。"
             exit 1
           fi
         else
           echo "未找到 blogs.json，跳过删除。"
         fi
     
         echo "等待 2 秒后开始构建..."
         sleep 2
         hugo --minify


     - name: Deploy
       if: steps.push_changes.outputs.push_success == 'true'
       uses: peaceiris/actions-gh-pages@v3
       with:
         personal_token: ${{ secrets.MY_GITHUB_TOKEN }}   
         external_repository: rvnox/rvnox.github.io    
         publish_dir: ./public
         publish_branch: main
         user_name: 'github-actions[bot]'
         user_email: 'github-actions[bot]@users.noreply.github.com'
